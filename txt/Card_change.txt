//Card.jsx
import { useState } from "react";
import './Card.css'

function Card({num, state}){
    const [color, setColor]=useState('white');

    const reserveColor=['white', 'green', 'red']

    const changeColor=() => {
        const currentIndex=reserveColor.indexOf(color);

        let nextIndex=currentIndex+1;

        if(nextIndex === reserveColor.length){
            nextIndex=0;
        }
        
        setColor(reserveColor[nextIndex]);
    }

    let textcolor='black';

    if(color === 'green'){
        state='선택됨'
        textcolor='white'
    }
    else if(color === 'red'){
        state='예매 완료'
        textcolor='white'
    }
    else(state='예매 가능')

    return(
        <div className="card" onClick={changeColor} style={{backgroundColor: color, color: textcolor}}>
            <div className="num" style={{color: textcolor}}>{num}</div>
            <div className="state">{state}</div>
        </div>
    )
}

export default Card;
